Once you have the app installed, the consent banner will be shown. If you're looking for a place to show your cookie policy, active cookies, and changes to consent, you can render the cookie declaration page in your Vue application.

``` vue [App.vue]
<script setup lang="ts">
  // Vendor
  import { useCookiebot } from '@ambitiondev/vue-cookiebot';
  import { onMounted, useTemplateRef } from 'vue';

  // Refs
  const cookieDeclarationRef = useTemplateRef('cookie-declaration');

  // Composable
  const { cookieDeclaration } = useCookiebot();

  onMounted(() => {
    cookieDeclaration(cookieDeclarationRef);
  });
</script>

<template>
    <div ref="cookie-declaration"></div>
</template>
```
